cmake_minimum_required(VERSION 3.25)
project(KekLocatorWindowsTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
get_filename_component(KEK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(KEK_SDK "${KEK_ROOT}/.tools/iplug2/Dependencies/IPlug/VST3_SDK")
add_subdirectory("${KEK_SDK}" vst-sdk)
if(NOT TARGET validator)
  message(FATAL_ERROR "The pinned SDK did not provide its validator target")
endif()
add_library(kek-rust STATIC IMPORTED)
set_target_properties(kek-rust PROPERTIES IMPORTED_LOCATION "${KEK_ROOT}/target/x86_64-pc-windows-msvc/release/keklocator_plugin.lib")
foreach(format clap vst3)
  if(format STREQUAL "clap")
    set(test_source "${KEK_ROOT}/tools/plugin-host-tests.mm")
  else()
    set(test_source "${KEK_ROOT}/tools/plugin-vst3-host-tests.mm")
  endif()
  set_source_files_properties("${test_source}" PROPERTIES LANGUAGE CXX)
  add_executable(kek-${format}-test "${test_source}")
  target_compile_options(kek-${format}-test PRIVATE /TP)
  target_compile_definitions(kek-${format}-test PRIVATE NOMINMAX)
  target_include_directories(kek-${format}-test PRIVATE "${KEK_SDK}" "${KEK_ROOT}/.tools/iplug2/Dependencies/IPlug/CLAP_SDK/include")
  target_link_libraries(kek-${format}-test PRIVATE kek-rust ws2_32 userenv bcrypt ntdll)
endforeach()
target_sources(kek-vst3-test PRIVATE "${KEK_SDK}/public.sdk/source/common/memorystream.cpp")
target_link_libraries(kek-vst3-test PRIVATE sdk_hosting sdk_common base pluginterfaces)
