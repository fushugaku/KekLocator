cmake_minimum_required(VERSION 3.22)
project(KekLocator VERSION 1.2.0 LANGUAGES C CXX)
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/iplug2" CACHE PATH "iPlug2 SDK")
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "Install with tools/install-plugins.sh" FORCE)
include(${IPLUG2_DIR}/iPlug2.cmake)
find_package(iPlug2 REQUIRED)
set(ECHO_RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../target/release/libkeklocator_plugin.a" CACHE FILEPATH "Rust DSP static library")
add_library(echo-rust STATIC IMPORTED)
set_target_properties(echo-rust PROPERTIES IMPORTED_LOCATION "${ECHO_RUST_LIBRARY}")
file(GLOB_RECURSE ECHO_WEB CONFIGURE_DEPENDS "resources/web/*")
set(ECHO_FORMATS VST3 CLAP)
if(APPLE)
  list(APPEND ECHO_FORMATS AU)
endif()
if(EXISTS "${IPLUG2_DIR}/Dependencies/IPlug/AAX_SDK/Interfaces/AAX.h")
  list(APPEND ECHO_FORMATS AAX)
endif()
iplug_add_plugin(KekLocator SOURCES KekLocator.cpp KekLocator.h engine.h
  WEB_RESOURCES ${ECHO_WEB} UI WEBVIEW FORMATS ${ECHO_FORMATS} LINK echo-rust)
if(WIN32)
  foreach(format vst3 clap)
    target_link_libraries(KekLocator-${format} PRIVATE ws2_32 userenv bcrypt ntdll)
  endforeach()
endif()
if(APPLE)
  foreach(format vst3 clap au)
    target_link_libraries(KekLocator-${format} PRIVATE "-framework Security" "-framework SystemConfiguration" "-framework CoreFoundation" "-liconv")
  endforeach()
endif()
